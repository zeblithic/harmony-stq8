// stq8-core/src/q8.rs

use serde::{Deserialize, Serialize};

const CONSONANTS: [char; 4] = ['\'', 'J', 'K', 'V'];
const VOWELS: [char; 4] = ['O', 'U', 'E', 'I'];

/// The 4 Q8 consonants (high 2 bits of a nibble).
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum Consonant {
    GlottalStop, // 00
    J,           // 01
    K,           // 10
    V,           // 11
}

/// The 4 Q8 vowels (low 2 bits of a nibble).
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum Vowel {
    O, // 00
    U, // 01
    E, // 10
    I, // 11
}

/// One of the 8 Q8 phonemes.
#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
pub enum Phoneme {
    Consonant(Consonant),
    Vowel(Vowel),
}

impl Consonant {
    pub fn bits(self) -> u8 {
        match self {
            Self::GlottalStop => 0b00,
            Self::J => 0b01,
            Self::K => 0b10,
            Self::V => 0b11,
        }
    }

    pub fn from_bits(b: u8) -> Self {
        match b & 0b11 {
            0b00 => Self::GlottalStop,
            0b01 => Self::J,
            0b10 => Self::K,
            0b11 => Self::V,
            _ => unreachable!(),
        }
    }

    pub fn char(self) -> char {
        CONSONANTS[self.bits() as usize]
    }
}

impl Vowel {
    pub fn bits(self) -> u8 {
        match self {
            Self::O => 0b00,
            Self::U => 0b01,
            Self::E => 0b10,
            Self::I => 0b11,
        }
    }

    pub fn from_bits(b: u8) -> Self {
        match b & 0b11 {
            0b00 => Self::O,
            0b01 => Self::U,
            0b10 => Self::E,
            0b11 => Self::I,
            _ => unreachable!(),
        }
    }

    pub fn char(self) -> char {
        VOWELS[self.bits() as usize]
    }
}

/// Convert a 4-bit nibble to a two-character syllable string.
pub fn nibble_to_syllable(nibble: u8) -> String {
    let consonant = Consonant::from_bits((nibble >> 2) & 0x03);
    let vowel = Vowel::from_bits(nibble & 0x03);
    format!("{}{}", consonant.char(), vowel.char())
}

/// Parse a two-character syllable string back to a 4-bit nibble.
pub fn syllable_to_nibble(s: &str) -> Option<u8> {
    let mut chars = s.chars();
    let c = chars.next()?;
    let v = chars.next()?;
    if chars.next().is_some() {
        return None; // too long
    }

    let consonant = CONSONANTS.iter().position(|&x| x == c)? as u8;
    let vowel = VOWELS.iter().position(|&x| x == v)? as u8;
    Some((consonant << 2) | vowel)
}

/// Convert a byte to a 4-character Q8 word (two syllables).
pub fn byte_to_word(b: u8) -> String {
    let high = (b >> 4) & 0x0F;
    let low = b & 0x0F;
    format!("{}{}", nibble_to_syllable(high), nibble_to_syllable(low))
}

/// Parse a 4-character Q8 word back to a byte.
pub fn word_to_byte(word: &str) -> Option<u8> {
    if word.len() != 4 {
        return None;
    }
    let high = syllable_to_nibble(&word[0..2])?;
    let low = syllable_to_nibble(&word[2..4])?;
    Some((high << 4) | low)
}

/// Encode a byte slice into Q8 text (8 words per line, space-separated).
pub fn encode(data: &[u8]) -> String {
    let words: Vec<String> = data.iter().map(|&b| byte_to_word(b)).collect();
    let mut lines = Vec::new();
    for chunk in words.chunks(8) {
        lines.push(chunk.join(" "));
    }
    lines.join("\n")
}

/// Decode Q8 text back to bytes.
pub fn decode(text: &str) -> Result<Vec<u8>, DecodeError> {
    let mut bytes = Vec::new();
    for word in text.split_whitespace() {
        bytes.push(word_to_byte(word).ok_or(DecodeError::InvalidWord(word.to_string()))?);
    }
    Ok(bytes)
}

/// Error when decoding Q8 text.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidWord(String),
}

impl core::fmt::Display for DecodeError {
    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
        match self {
            Self::InvalidWord(w) => write!(f, "invalid Q8 word: {w}"),
        }
    }
}

**Context:** MFCCs (Mel-Frequency Cepstral Coefficients) are the standard feature representation for speech audio. The pipeline is: raw PCM → pre-emphasis → windowing → FFT → mel filterbank → log → DCT → coefficients. For our 8-phoneme classifier, we extract 13 MFCCs per frame, then summarize across all frames in a syllable using mean and variance (26 floats). With delta coefficients, that's 52 floats total per syllable.

**Reference:** The input is `&[f32]` PCM samples at 16kHz. Frame size is 400 samples (25ms). Hop size is 160 samples (10ms).

Implementation notes:
- `hz_to_mel`: `2595.0 * (1.0 + hz / 700.0).log10()`
- `mel_to_hz`: `700.0 * (10.0_f32.powf(mel / 2595.0) - 1.0)`
- For `extract_frame_mfccs`: pre-emphasize → window → zero-pad to FFT_SIZE → FFT → power spectrum → mel filterbank → log (with floor of 1e-10 to avoid log(0)) → DCT (type-II, first 13 coefficients)
- For `extract_features`: split signal into overlapping frames (FRAME_LEN, HOP_LEN) → extract MFCCs per frame → compute delta coefficients (first difference between adjacent frames) → compute mean and variance of static MFCCs (13 × 2 = 26) and delta MFCCs (13 × 2 = 26) → concatenate to 52 floats
- Use `rustfft::FftPlanner` for the FFT. Create the planner once and reuse.
- DCT type-II: `c[k] = sum_n(x[n] * cos(PI * k * (2n + 1) / (2 * N)))` for k = 0..NUM_MFCC

**Step 3: Implement segmenter.rs**

Implement `pub fn segment(samples: &[f32], config: &SegmenterConfig) -> Vec<SyllableBounds>`:

Algorithm:
1. Compute RMS energy per analysis frame (use FRAME_LEN and HOP_LEN from mfcc module, or use a simpler fixed window like 160 samples / 10ms).
2. Estimate noise floor from first `noise_floor_frames` frames (mean RMS of initial silence before PTT audio arrives — or minimum RMS across all frames if no leading silence).
3. Mark frames where RMS > noise_floor × onset_threshold as "active."
4. Find contiguous runs of active frames → candidate syllables.
5. Merge candidates separated by fewer than `min_gap` samples.
6. Reject candidates shorter than `min_duration` samples.
7. Return `Vec<SyllableBounds>` with start/end sample indices.

Also implement `Default` for `SegmenterConfig` matching the test's `default_config()`.

use crate::q8::Phoneme;
use crate::mfcc::FEATURE_DIM;

/// Result of classifying a single phoneme.
#[derive(Debug, Clone)]
pub struct Classification {
    pub phoneme: Phoneme,
    pub confidence: f32,
}

/// Classifier trait — abstracts inference backend.
pub trait Classifier {
    /// Classify a feature vector into a phoneme with confidence.
    fn classify(&self, features: &[f32]) -> Option<Classification>;

    /// Train/calibrate from labeled samples.
    fn train(&mut self, samples: &[(Phoneme, Vec<f32>)]);

    /// Whether the classifier has been trained.
    fn is_trained(&self) -> bool;
}

#[cfg(test)]
mod tests {
    use super::*;
    use crate::q8::{Consonant, Vowel};

    /// Make a synthetic feature vector centered at a given "position."
    /// Each phoneme gets a distinct region in feature space.
    fn make_features(phoneme_idx: usize) -> Vec<f32> {
        let mut v = vec![0.0f32; FEATURE_DIM];
        // Set a few dimensions to phoneme-specific values
        // so each phoneme is clearly separable
        v[phoneme_idx % FEATURE_DIM] = 1.0;
        v[(phoneme_idx * 7) % FEATURE_DIM] = 0.5;
        v[(phoneme_idx * 13) % FEATURE_DIM] = 0.8;
        v
    }

    fn all_phonemes() -> Vec<Phoneme> {
        vec![
            Phoneme::Consonant(Consonant::GlottalStop),
            Phoneme::Consonant(Consonant::J),
            Phoneme::Consonant(Consonant::K),
            Phoneme::Consonant(Consonant::V),
            Phoneme::Vowel(Vowel::O),
            Phoneme::Vowel(Vowel::U),
            Phoneme::Vowel(Vowel::E),
            Phoneme::Vowel(Vowel::I),
        ]
    }

    fn train_with_synthetic(classifier: &mut impl Classifier) {
        let phonemes = all_phonemes();
        let mut samples = Vec::new();
        for (i, p) in phonemes.iter().enumerate() {
            // 5 samples per phoneme with slight noise
            for offset in 0..5 {
                let mut feats = make_features(i);
                feats[offset % FEATURE_DIM] += 0.01 * offset as f32;
                samples.push((*p, feats));
            }
        }
        classifier.train(&samples);
    }

    #[test]
    fn untrained_returns_none() {
        let c = NearestCentroid::new();
        assert!(!c.is_trained());
        assert!(c.classify(&vec![0.0; FEATURE_DIM]).is_none());
    }

    #[test]
    fn trained_classifies_correctly() {
        let mut c = NearestCentroid::new();
        train_with_synthetic(&mut c);
        assert!(c.is_trained());

        let phonemes = all_phonemes();
        for (i, p) in phonemes.iter().enumerate() {
            let feats = make_features(i);
            let result = c.classify(&feats).expect("should classify");
            assert_eq!(result.phoneme, *p, "phoneme {i} misclassified");
            assert!(result.confidence > 0.5, "confidence too low: {}", result.confidence);
        }
    }

    #[test]
    fn exact_centroid_high_confidence() {
        let mut c = NearestCentroid::new();
        train_with_synthetic(&mut c);

        let feats = make_features(0); // exact match for phoneme 0
        let result = c.classify(&feats).unwrap();
        assert!(result.confidence > 0.8, "exact match should have high confidence: {}", result.confidence);
    }

    #[test]
    fn equidistant_low_confidence() {
        let mut c = NearestCentroid::new();
        train_with_synthetic(&mut c);

        // Zero vector is equidistant from everything
        let feats = vec![0.0; FEATURE_DIM];
        let result = c.classify(&feats).unwrap();
        assert!(result.confidence < 0.5, "equidistant should have low confidence: {}", result.confidence);
    }

    #[test]
    fn wrong_feature_dim_returns_none() {
        let mut c = NearestCentroid::new();
        train_with_synthetic(&mut c);
        assert!(c.classify(&[1.0, 2.0, 3.0]).is_none()); // wrong dimension

```rust
pub struct NearestCentroid {
    centroids: Vec<(Phoneme, Vec<f32>)>,
}
```

- `new()`: empty, untrained
- `train()`: compute mean feature vector per phoneme from all samples of that phoneme
- `classify()`: compute cosine similarity against all centroids, return best match. Confidence = `1.0 - (second_best_similarity / best_similarity)` clamped to [0, 1]. If features length != FEATURE_DIM, return None.

Cosine similarity: `dot(a, b) / (norm(a) * norm(b))`. Handle zero-norm vectors gracefully (return 0.0).

**Context:** A `Tome` is a personal spellbook of `Scroll` entries, addressed by Q8 nibble sequences. It uses a trie for shortest-prefix resolution (say one syllable, get a unique result if possible) and power-of-two-choices for collision avoidance when inserting.

/// Maximum scroll payload size in bytes.

/// What a scroll contains.
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub enum ScrollKind {
    /// A text note.
    Text(String),
    /// Reference to content by CID string.
    Reference(String),
    /// Redirect to another Q8 address.
    Alias(Vec<u8>),

    Action { program: String, args: Vec<u8> },
}

    pub kind: ScrollKind,

    pub created_epoch_secs: u64,
    pub updated_epoch_secs: u64,
}

use stq8_core::pipeline::{Pipeline, UtteranceResult};
use stq8_core::q8::{self, Phoneme, Consonant, Vowel};
use stq8_core::mfcc;

#[wasm_bindgen]
pub struct WasmPipeline {

}

#[wasm_bindgen]
impl WasmPipeline {
    #[wasm_bindgen(constructor)]
    pub fn new() -> Self {
        Self { inner: Pipeline::new() }
    }

    /// Feed calibration samples. `phoneme_index` is 0-7 (4 consonants then 4 vowels).
    /// `pcm` is raw f32 samples at 16kHz.
    pub fn add_calibration_sample(&mut self, phoneme_index: u8, pcm: &[f32]) {
        let phoneme = index_to_phoneme(phoneme_index);
        let features = mfcc::extract_features(pcm);
        // Store for batch training — accumulate then call finalize_calibration
        self.inner.add_calibration_sample(phoneme, features);
    }

    /// Finalize calibration after all samples are added.
    pub fn finalize_calibration(&mut self) {
        self.inner.finalize_calibration();
    }

    pub fn is_calibrated(&self) -> bool {
        self.inner.is_calibrated()
    }

    /// Process a PTT utterance. Returns JSON-encoded UtteranceResult.
    pub fn process(&self, pcm: &[f32]) -> String {
        let result = self.inner.process(pcm);
        serde_json::to_string(&result).unwrap_or_default()
    }

    /// Encode bytes to Q8 text.
    pub fn encode_q8(data: &[u8]) -> String {
        q8::encode(data)
    }

    /// Decode Q8 text to bytes. Returns empty vec on error.
    pub fn decode_q8(text: &str) -> Vec<u8> {
        q8::decode(text).unwrap_or_default()
    }

    pub fn export_profile(&self) -> String {
        self.inner.export_profile_json().unwrap_or_default()
    }

    pub fn import_profile(&mut self, json: &str) -> bool {
        self.inner.import_profile_json(json).is_ok()
    }

fn index_to_phoneme(idx: u8) -> Phoneme {
    match idx {
        0 => Phoneme::Consonant(Consonant::GlottalStop),
        1 => Phoneme::Consonant(Consonant::J),
        2 => Phoneme::Consonant(Consonant::K),
        3 => Phoneme::Consonant(Consonant::V),
        4 => Phoneme::Vowel(Vowel::O),

        7 => Phoneme::Vowel(Vowel::I),
        _ => Phoneme::Vowel(Vowel::O), // fallback
    }
}
